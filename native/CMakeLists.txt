cmake_minimum_required(VERSION 3.21)
project(AgoraNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network Sql Multimedia Quick Qml Test QuickTest)
find_package(OpenSSL REQUIRED)
if(DEFINED AGORA_EXPECTED_QT_VERSION AND NOT Qt6_VERSION VERSION_EQUAL AGORA_EXPECTED_QT_VERSION)
    message(FATAL_ERROR "Expected Qt ${AGORA_EXPECTED_QT_VERSION}, found ${Qt6_VERSION}")
endif()
set(backend_sources backend/ForumBackend.cpp backend/ForumBackend.h
    backend/VideoSession.cpp backend/VideoSession.h backend/StorageTransfer.cpp
    backend/StorageTransfer.h backend/AttachmentMime.h)
set(backend_libraries Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql Qt6::Multimedia OpenSSL::Crypto)
add_library(forum_backend MODULE ${backend_sources})
target_link_libraries(forum_backend PRIVATE ${backend_libraries})
add_library(agoramedia MODULE media-renderer/FrameSurface.cpp)
target_link_libraries(agoramedia PRIVATE Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml)
if(WIN32)
    set_target_properties(forum_backend agoramedia PROPERTIES PREFIX "")
endif()
add_executable(backend_tests tests/backend_tests.cpp ${backend_sources})
target_link_libraries(backend_tests PRIVATE ${backend_libraries} Qt6::Test)
add_executable(forum-ui-tests tests/ui_main.cpp ${backend_sources})
target_link_libraries(forum-ui-tests PRIVATE ${backend_libraries} Qt6::Quick Qt6::Qml Qt6::QuickTest)
set(renderer_dir "build/qt${Qt6_VERSION}-media/qml/Agora/Media")
install(TARGETS forum_backend RUNTIME DESTINATION native/forum LIBRARY DESTINATION native/forum)
install(TARGETS agoramedia RUNTIME DESTINATION ${renderer_dir} LIBRARY DESTINATION ${renderer_dir})
install(FILES media-renderer/qmldir DESTINATION ${renderer_dir})
install(TARGETS backend_tests forum-ui-tests RUNTIME DESTINATION native/tests)
